# ExoDiscover ML Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a leakage-free, star-grouped, calibrated exoplanet classifier over the NASA Kepler KOI catalog that emits a versioned model artifact and a `metrics.json` consumable by the API and web layers.

**Architecture:** A `ml/exodiscover/` Python package with a strict separation between ingestion (NASA TAP API → `data/raw/`), feature construction (physics-informed transforms behind a leakage firewall), model selection (a ladder of classifiers tuned under `StratifiedGroupKFold` grouped on `kepid`), and evaluation (nested CV + a star-held-out test set, emitting JSON and plots). A Typer CLI drives each stage. Correctness claims are enforced by tests, not by inspection: a leakage guard test and a group-split test both fail CI if the invariants break.

**Tech Stack:** Python 3.11, pandas, numpy, scikit-learn, XGBoost, LightGBM, CatBoost, Optuna, SHAP, Typer, pydantic-settings, pytest, ruff, mypy.

**Spec:** `docs/superpowers/specs/2026-08-30-exodiscover-overhaul-design.md`

## Global Constraints

- **Python 3.11.** Declared as `requires-python = ">=3.11,<3.13"` in `pyproject.toml`.
- **CPU only.** No CUDA dependencies. No TensorFlow, no PyTorch. Total training budget is under one hour on a laptop CPU.
- **No network access at test time.** Every test runs against `tests/fixtures/koi_sample.csv`. Only `exo ingest` may contact the NASA archive.
- **Global seed is 42**, defined once in `ml/exodiscover/config.py` as `Settings.random_seed` and threaded into every estimator and split. No module may hardcode a seed literal.
- **Grouping column is `kepid`.** Every split in the project is grouped on it. There are no exceptions.
- **`LEAKY_COLUMNS` is the single source of truth** for forbidden features. No module may define its own list.
- **Cross-platform paths.** Use `pathlib.Path` throughout; no backslash string literals, no absolute paths.
- **Do not run `git commit` or `git push`.** The project owner commits their own work. Where a task ends in "Commit", stage nothing — instead print the suggested message and stop for the owner.
- **Artifacts are gitignored** except `models/production/` and `docs/metrics/metrics.json`.

```python
"""Column contracts for the NASA exoplanet catalogs.

LEAKY_COLUMNS is the single source of truth for features that encode the
target. koi_score is the Robovetter's own disposition score and the
koi_fpflag_* family are its false-positive vetting decisions; a model given
any of them reproduces the label rather than learning the physics.
"""

from __future__ import annotations

import pandas as pd

LEAKY_COLUMNS: frozenset[str] = frozenset(
    {
        "koi_disposition",
        "koi_pdisposition",
        "koi_score",
        "koi_fpflag_nt",
        "koi_fpflag_ss",
        "koi_fpflag_co",
        "koi_fpflag_ec",
        "koi_tce_delivname",
        "tfopwg_disp",
        "disposition",
    }
)

KOI_LABEL_MAP: dict[str, int] = {
    "FALSE POSITIVE": 0,
    "CANDIDATE": 1,
    "CONFIRMED": 2,
}

# TESS Follow-up Working Group dispositions.
# CP = Confirmed Planet, KP = Known Planet (also confirmed), PC = Planet
# Candidate, APC = Ambiguous Planet Candidate, FP = False Positive,
# FA = False Alarm.
TOI_LABEL_MAP: dict[str, int] = {
    "CP": 2,
    "KP": 2,
    "PC": 1,
    "APC": 1,
    "FP": 0,
    "FA": 0,
}


class LeakageError(ValueError):
    """Raised when a target-encoding column reaches a feature matrix."""


def assert_no_leakage(df: pd.DataFrame, *, context: str = "dataframe") -> None:
    offenders = sorted(set(df.columns) & LEAKY_COLUMNS)
    if offenders:
        raise LeakageError(
            f"{len(offenders)} leaky column(s) present in {context}: "
            f"{', '.join(offenders)}. These encode the target; see docs/LEAKAGE.md."
        )
```

```python
"""Client for the NASA Exoplanet Archive TAP service.

Replaces the four ad-hoc download scripts in the original backend. Every
fetch records its query text and timestamp so a training run can state
exactly which archive snapshot produced it.
"""

from __future__ import annotations

import json
from datetime import UTC, datetime
from io import StringIO

import pandas as pd
import requests

from exodiscover.config import settings

TAP_URL = "https://exoplanetarchive.ipac.caltech.edu/TAP/sync"

TABLES: dict[str, str] = {
    "koi": "cumulative",
    "toi": "toi",
    "k2": "k2pandc",
}


def build_query(table: str) -> str:
    if table not in TABLES:
        raise KeyError(f"unknown table {table!r}; expected one of {sorted(TABLES)}")
    return f"select * from {TABLES[table]}"


def _download(query: str) -> pd.DataFrame:
    response = requests.get(
        TAP_URL,
        params={"query": query, "format": "csv"},
        timeout=300,
    )
    response.raise_for_status()
    return pd.read_csv(StringIO(response.text))


def _paths(table: str) -> tuple:
    raw = settings.root / "data" / "raw"
    return raw / f"{table}.csv", raw / f"{table}.meta.json"


def fetch(table: str, *, force: bool = False) -> pd.DataFrame:
    csv_path, meta_path = _paths(table)
    if csv_path.exists() and not force:
        return pd.read_csv(csv_path)

    query = build_query(table)
    df = _download(query)

    csv_path.parent.mkdir(parents=True, exist_ok=True)
    df.to_csv(csv_path, index=False)
    meta_path.write_text(
        json.dumps(
            {
                "query": query,
                "fetched_at": datetime.now(UTC).isoformat(),
                "n_rows": int(len(df)),
            },
            indent=2,
        )
    )
    return df


def load_cached(table: str) -> pd.DataFrame:
    csv_path, _ = _paths(table)
    if not csv_path.exists():
        raise FileNotFoundError(f"{csv_path} missing; run `exo ingest` first")
    return pd.read_csv(csv_path)
```

```python
import pandas as pd

from exodiscover.data.splits import grouped_train_test_split, make_cv


def _frame(n_stars: int = 200) -> tuple:
    rng = np.random.default_rng(0)
    rows = []
    for star in range(n_stars):
        for _ in range(rng.integers(1, 4)):
            rows.append({"kepid": star, "feat": rng.normal(), "y": int(star % 2)})
    df = pd.DataFrame(rows)
    return df[["feat"]], df["y"], df["kepid"]


def test_no_star_appears_in_both_train_and_test():
    X, y, groups = _frame()
    X_tr, X_te, _, _ = grouped_train_test_split(X, y, groups, test_size=0.2, seed=42)
    train_stars = set(groups.loc[X_tr.index])
    test_stars = set(groups.loc[X_te.index])
    assert train_stars & test_stars == set(), "a kepid spans the split"


def test_split_is_deterministic_under_a_fixed_seed():
    X, y, groups = _frame()
    a = grouped_train_test_split(X, y, groups, seed=42)[1].index.tolist()
    b = grouped_train_test_split(X, y, groups, seed=42)[1].index.tolist()
```

`ml/exodiscover/data/splits.py`:

```python
"""Star-grouped splitting.

Kepler holds 9,564 KOI rows across only 8,214 stars, with up to seven KOIs
on one star. A random row split places sibling KOIs from the same star on
both sides, so every split in this project is grouped on kepid.
"""

from __future__ import annotations

import pandas as pd
from sklearn.model_selection import StratifiedGroupKFold

from exodiscover.config import settings


def make_cv(n_splits: int | None = None) -> StratifiedGroupKFold:
    return StratifiedGroupKFold(
        n_splits=n_splits or settings.cv_folds,
        shuffle=True,
        random_state=settings.random_seed,
    )


def grouped_train_test_split(
    X: pd.DataFrame,
    y: pd.Series,
    groups: pd.Series,
    *,
    test_size: float = 0.2,
    seed: int | None = None,
) -> tuple[pd.DataFrame, pd.DataFrame, pd.Series, pd.Series]:
    """Hold out whole stars. Implemented as one fold of a grouped K-fold so
    that stratification and grouping are honoured simultaneously."""
    n_splits = max(2, round(1 / test_size))
    splitter = StratifiedGroupKFold(
        n_splits=n_splits, shuffle=True, random_state=seed or settings.random_seed
    )
    train_idx, test_idx = next(splitter.split(X, y, groups))
    return X.iloc[train_idx], X.iloc[test_idx], y.iloc[train_idx], y.iloc[test_idx]
```

```python
"""Physics-informed features for KOI classification.

The discriminative power here comes from consistency checks rather than raw
values. A genuine planetary transit must satisfy relationships between
period, stellar density, duration, and depth; eclipsing binaries and blends
violate them in characteristic ways.
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from exodiscover.data.schema import LEAKY_COLUMNS, assert_no_leakage

G_CGS = 6.674e-8          # cm^3 g^-1 s^-2
R_SUN_CM = 6.957e10
R_EARTH_CM = 6.371e8
SEC_PER_DAY = 86_400.0

FEATURE_COLUMNS: list[str] = [
    "log_period",
    "log_depth",
    "log_prad",
    "log_insol",
    "koi_impact",
    "koi_model_snr",
    "koi_teq",
    "koi_steff",
    "koi_slogg",
    "koi_srad",
    "koi_kepmag",
    "koi_tce_plnt_num",
    "rho_star",
    "a_over_rstar",
    "duration_ratio",
    "depth_ratio",
    "rel_err_period",
    "rel_err_depth",
    "rel_err_duration",
    "n_kois_on_star",
]


def stellar_density(logg_cgs: np.ndarray, srad_rsun: np.ndarray) -> np.ndarray:
    """rho = 3g / (4 pi G R), from g = GM/R^2 and rho = M / (4/3 pi R^3)."""
    g = np.power(10.0, logg_cgs)
    r = srad_rsun * R_SUN_CM
    return 3.0 * g / (4.0 * np.pi * G_CGS * r)


def a_over_rstar(period_days: np.ndarray, rho_star: np.ndarray) -> np.ndarray:
    """Kepler's third law in the form a/R* = (G rho P^2 / 3 pi)^(1/3)."""
    p = period_days * SEC_PER_DAY
    return np.cbrt(G_CGS * rho_star * p**2 / (3.0 * np.pi))


def expected_duration_hours(
    period_days: np.ndarray, rho_star: np.ndarray, impact: np.ndarray
) -> np.ndarray:
    """Small-angle transit duration, T = (P / pi) * sqrt(1 - b^2) / (a/R*).

    A grazing or impossible geometry (b >= 1) returns 0 rather than NaN.
    """
    a_r = a_over_rstar(period_days, rho_star)
    chord = np.sqrt(np.clip(1.0 - impact**2, 0.0, None))
    return period_days * 24.0 / (np.pi * a_r) * chord


def expected_depth_ppm(prad_earth: np.ndarray, srad_rsun: np.ndarray) -> np.ndarray:
    """(Rp / R*)^2 expressed in parts per million."""
    k = (prad_earth * R_EARTH_CM) / (srad_rsun * R_SUN_CM)
    return k**2 * 1e6


def _safe_log10(x: pd.Series) -> pd.Series:
    return np.log10(x.clip(lower=1e-9))


def _safe_ratio(num: pd.Series, den: pd.Series) -> pd.Series:
    return (num / den.replace(0.0, np.nan)).replace([np.inf, -np.inf], np.nan)


def build_features(koi: pd.DataFrame, *, _skip_drop: bool = False) -> pd.DataFrame:
    """Build the model feature matrix from a raw KOI frame.

    Leaky columns are dropped up front, then the result is asserted clean so
    an engineered feature can never smuggle one back in.
    """
    df = koi if _skip_drop else koi.drop(columns=list(LEAKY_COLUMNS), errors="ignore")
    out = pd.DataFrame(index=df.index)

    out["log_period"] = _safe_log10(df["koi_period"])
    out["log_depth"] = _safe_log10(df["koi_depth"])
    out["log_prad"] = _safe_log10(df["koi_prad"])
    out["log_insol"] = _safe_log10(df.get("koi_insol", pd.Series(1.0, index=df.index)))

    for col in [
        "koi_impact", "koi_model_snr", "koi_teq", "koi_steff",
        "koi_slogg", "koi_srad", "koi_kepmag", "koi_tce_plnt_num",
    ]:
        out[col] = pd.to_numeric(df.get(col), errors="coerce")

    rho = stellar_density(out["koi_slogg"].to_numpy(), out["koi_srad"].to_numpy())
    out["rho_star"] = rho
    out["a_over_rstar"] = a_over_rstar(df["koi_period"].to_numpy(), rho)

    expected_dur = expected_duration_hours(
        df["koi_period"].to_numpy(), rho, out["koi_impact"].fillna(0.0).to_numpy()
    )
    out["duration_ratio"] = _safe_ratio(
        df["koi_duration"], pd.Series(expected_dur, index=df.index)
    )

    expected_depth = expected_depth_ppm(
        df["koi_prad"].to_numpy(), out["koi_srad"].to_numpy()
    )
    out["depth_ratio"] = _safe_ratio(df["koi_depth"], pd.Series(expected_depth, index=df.index))

    out["rel_err_period"] = _safe_ratio(df["koi_period_err1"].abs(), df["koi_period"])
    out["rel_err_depth"] = _safe_ratio(df["koi_depth_err1"].abs(), df["koi_depth"])
    out["rel_err_duration"] = _safe_ratio(df["koi_duration_err1"].abs(), df["koi_duration"])

    out["n_kois_on_star"] = df.groupby("kepid")["kepid"].transform("size")

    out = out[FEATURE_COLUMNS]
    out = out.replace([np.inf, -np.inf], np.nan)
    out = out.fillna(out.median(numeric_only=True)).fillna(0.0)

    assert_no_leakage(out, context="feature matrix")
    return out
```

Note the two-sided defence: leaky columns are dropped on the way in, and `assert_no_leakage` re-checks on the way out, so an engineered feature can never reintroduce one under a surviving name.

```python
from __future__ import annotations

from dataclasses import dataclass, field

import numpy as np
import optuna
import pandas as pd
from sklearn.base import BaseEstimator, clone
from sklearn.calibration import CalibratedClassifierCV
from sklearn.model_selection import cross_validate

from exodiscover.config import settings
from exodiscover.data.splits import make_cv
from exodiscover.models.registry import candidates

SCORING = {"pr_auc": "average_precision", "roc_auc": "roc_auc", "brier": "neg_brier_score"}


@dataclass
class TrainResult:
    name: str
    estimator: BaseEstimator
    cv_scores: dict[str, float] = field(default_factory=dict)


def _score(estimator, X, y, groups) -> dict[str, float]:
    out = cross_validate(
        estimator, X, y, groups=groups, cv=make_cv(), scoring=SCORING, n_jobs=1
    )
    return {k: float(np.mean(out[f"test_{k}"])) for k in SCORING}


def fit_candidates(X: pd.DataFrame, y: pd.Series, groups: pd.Series) -> list[TrainResult]:
    results = [
        TrainResult(name=name, estimator=est, cv_scores=_score(est, X, y, groups))
        for name, est in candidates(settings.random_seed).items()
    ]
    return sorted(results, key=lambda r: r.cv_scores["pr_auc"], reverse=True)


def tune_best(X, y, groups, name: str, n_trials: int = 50) -> BaseEstimator:
    """Optuna search over the winning family. Only the tree families are
    tuned; logreg and dummy have nothing worth searching."""
    optuna.logging.set_verbosity(optuna.logging.WARNING)
    base = candidates(settings.random_seed)[name]

    def objective(trial: optuna.Trial) -> float:
        params = {
            "max_depth": trial.suggest_int("max_depth", 3, 9),
            "learning_rate": trial.suggest_float("learning_rate", 0.01, 0.2, log=True),
            "n_estimators": trial.suggest_int("n_estimators", 200, 800, step=100),
        }
        est = clone(base)
        est.set_params(**{k: v for k, v in params.items() if k in est.get_params()})
        return _score(est, X, y, groups)["pr_auc"]

    study = optuna.create_study(
        direction="maximize",
        sampler=optuna.samplers.TPESampler(seed=settings.random_seed),
    )
    study.optimize(objective, n_trials=n_trials, show_progress_bar=False)

    best = clone(base)
    best.set_params(**{k: v for k, v in study.best_params.items() if k in best.get_params()})
    return best


def build_calibrated(estimator, X, y, groups) -> CalibratedClassifierCV:
    """Isotonic calibration under the same grouped CV. Calibration matters
    here because the discovery ranking consumes probabilities, not classes."""
    model = CalibratedClassifierCV(clone(estimator), method="isotonic", cv=make_cv())
    model.fit(X, y)
    return model
```

```python
import json
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402
import numpy as np  # noqa: E402
from sklearn.metrics import (  # noqa: E402
    average_precision_score,
    brier_score_loss,
    classification_report,
    confusion_matrix,
    precision_recall_curve,
    roc_auc_score,
)

from exodiscover.config import settings  # noqa: E402


def precision_at_k(y_true: np.ndarray, y_prob: np.ndarray, k: int = 50) -> float:
    """Fraction of the k highest-ranked items that are true positives.
    This is the metric a vetting team actually cares about: they work a
    finite shortlist, not the whole catalog."""
    k = min(k, len(y_true))
    top = np.argsort(y_prob)[::-1][:k]
    return float(np.mean(np.asarray(y_true)[top]))


def reliability_curve(y_true: np.ndarray, y_prob: np.ndarray, n_bins: int = 10) -> dict:
    edges = np.linspace(0.0, 1.0, n_bins + 1)
    idx = np.clip(np.digitize(y_prob, edges) - 1, 0, n_bins - 1)
    centers, observed, counts = [], [], []
    for b in range(n_bins):
        mask = idx == b
        if not mask.any():
            continue
        centers.append(float(np.mean(y_prob[mask])))
        observed.append(float(np.mean(np.asarray(y_true)[mask])))
        counts.append(int(mask.sum()))
    return {"bin_centers": centers, "observed": observed, "counts": counts}
```

```python
"""Quantifies the three failure modes found in the original pipeline:
label leakage, split leakage, and a task framing that encodes follow-up
selection rather than physics.
"""

from __future__ import annotations

import numpy as np
import pandas as pd
from sklearn.ensemble import HistGradientBoostingClassifier
from sklearn.model_selection import train_test_split

from exodiscover.config import settings
from exodiscover.data.splits import grouped_train_test_split
from exodiscover.evaluate import evaluate_binary
from exodiscover.features.tabular import build_features

LEAKY_EXTRAS = ["koi_score", "koi_fpflag_nt", "koi_fpflag_ss", "koi_fpflag_co", "koi_fpflag_ec"]


def collapse_to_planetlike(y: np.ndarray) -> np.ndarray:
    """Map 3-class predictions onto the planet-like vs false-positive
    decision that the binary framing also makes, so the two are comparable."""
    return (np.asarray(y) > 0).astype(int)


def _model() -> HistGradientBoostingClassifier:
    return HistGradientBoostingClassifier(random_state=settings.random_seed)


def _binary_frame(koi: pd.DataFrame) -> tuple:
    df = koi[koi["koi_disposition"].isin(["CONFIRMED", "FALSE POSITIVE"])].reset_index(drop=True)
    y = (df["koi_disposition"] == "CONFIRMED").astype(int)
    return df, y


def run_leakage_ablation(koi: pd.DataFrame) -> list[dict]:
    df, y = _binary_frame(koi)
    clean = build_features(df)
    leaky = clean.copy()
    for col in LEAKY_EXTRAS:
        if col in df.columns:
            leaky[col] = pd.to_numeric(df[col], errors="coerce").fillna(0.0)

    rows = []
    seed = settings.random_seed

    Xtr, Xte, ytr, yte = train_test_split(leaky, y, test_size=0.2, random_state=seed, stratify=y)
    m = _model().fit(Xtr, ytr)
    rows.append({"setup": "leaky_random_split", **_slim(evaluate_binary(m, Xte, yte)),
                 "note": "koi_score + fpflags present, rows split at random (original-style)"})

    Xtr, Xte, ytr, yte = train_test_split(clean, y, test_size=0.2, random_state=seed, stratify=y)
    m = _model().fit(Xtr, ytr)
    rows.append({"setup": "clean_random_split", **_slim(evaluate_binary(m, Xte, yte)),
                 "note": "leakage removed, but sibling KOIs still span the split"})

    Xtr, Xte, ytr, yte = grouped_train_test_split(clean, y, df["kepid"])
    m = _model().fit(Xtr, ytr)
    rows.append({"setup": "clean_grouped_split", **_slim(evaluate_binary(m, Xte, yte)),
                 "note": "leakage removed and whole stars held out — the honest number"})

    return rows


def _slim(metrics: dict) -> dict:
    return {k: metrics[k] for k in ("roc_auc", "pr_auc", "brier", "precision_at_50")}


def run_framing_ablation(koi: pd.DataFrame) -> list[dict]:
    rows = []

    # A: flat 3-class, scored on the collapsed planet-like decision.
    df3 = koi.dropna(subset=["koi_disposition"]).reset_index(drop=True)
    y3 = df3["koi_disposition"].map({"FALSE POSITIVE": 0, "CANDIDATE": 1, "CONFIRMED": 2})
    X3 = build_features(df3)
    Xtr, Xte, ytr, yte = grouped_train_test_split(X3, y3, df3["kepid"])
    m = _model().fit(Xtr, ytr)
    proba_planetlike = m.predict_proba(Xte)[:, 1:].sum(axis=1)
    rows.append({"framing": "3class",
                 **_binary_scores(collapse_to_planetlike(yte), proba_planetlike)})

    # B: binary confirmed vs false positive.
    dfb, yb = _binary_frame(koi)
    Xb = build_features(dfb)
    Xtr, Xte, ytr, yte = grouped_train_test_split(Xb, yb, dfb["kepid"])
    m = _model().fit(Xtr, ytr)
    rows.append({"framing": "binary",
                 **_binary_scores(yte, m.predict_proba(Xte)[:, 1])})

    # C: diagnostic — confirmed vs candidate among vetted signals. If this
    # separates well on koi_kepmag / multiplicity / period, the third class
    # is encoding follow-up selection rather than transit physics.
    dfd = koi[koi["koi_disposition"].isin(["CONFIRMED", "CANDIDATE"])].reset_index(drop=True)
    yd = (dfd["koi_disposition"] == "CONFIRMED").astype(int)
    Xd = build_features(dfd)
    Xtr, Xte, ytr, yte = grouped_train_test_split(Xd, yd, dfd["kepid"])
    m = _model().fit(Xtr, ytr)
    rows.append({"framing": "diagnostic",
                 **_binary_scores(yte, m.predict_proba(Xte)[:, 1])})

    return rows


def _binary_scores(y_true, y_prob) -> dict:
    from sklearn.metrics import average_precision_score, brier_score_loss, roc_auc_score

    return {
        "roc_auc": float(roc_auc_score(y_true, y_prob)),
        "pr_auc": float(average_precision_score(y_true, y_prob)),
        "brier": float(brier_score_loss(y_true, y_prob)),
    }
```

```python
from __future__ import annotations

import numpy as np
import pandas as pd
import shap
from sklearn.calibration import CalibratedClassifierCV


def unwrap(model):
    """SHAP needs the tree model, not the calibration wrapper around it."""
    if isinstance(model, CalibratedClassifierCV):
        return model.calibrated_classifiers_[0].estimator
    return model


def _shap_matrix(model, X: pd.DataFrame) -> np.ndarray:
    values = shap.TreeExplainer(unwrap(model)).shap_values(X)
    if isinstance(values, list):          # older API: one array per class
        values = values[1]
    elif values.ndim == 3:                # newer API: (n, features, classes)
        values = values[:, :, 1]
    return np.asarray(values)


def global_importance(model, X: pd.DataFrame) -> list[dict]:
    mean_abs = np.abs(_shap_matrix(model, X)).mean(axis=0)
    rows = [
        {"feature": f, "mean_abs_shap": float(v)}
        for f, v in zip(X.columns, mean_abs, strict=True)
    ]
    return sorted(rows, key=lambda r: r["mean_abs_shap"], reverse=True)


def explain_row(model, X: pd.DataFrame, index: int = 0, top_n: int = 6) -> list[dict]:
    row_shap = _shap_matrix(model, X.iloc[[index]])[0]
    rows = [
        {"feature": f, "value": float(X.iloc[index][f]), "shap": float(s)}
        for f, s in zip(X.columns, row_shap, strict=True)
    ]
    return sorted(rows, key=lambda r: abs(r["shap"]), reverse=True)[:top_n]
```

```python
"""Ranks unvetted KOI candidates by calibrated planet probability.

This is the project's actual output: a shortlist a vetting team could work,
each row carrying the reasons the model produced it.
"""

from __future__ import annotations

import pandas as pd

from exodiscover.explain import explain_row
from exodiscover.features.tabular import build_features


def rank_candidates(model, koi: pd.DataFrame, top_n: int = 50) -> pd.DataFrame:
    candidates_df = koi[koi["koi_disposition"] == "CANDIDATE"].reset_index(drop=True)
    if candidates_df.empty:
        return pd.DataFrame(
            columns=["kepoi_name", "kepler_name", "kepid", "probability", "rank", "top_reasons"]
        )

    X = build_features(candidates_df)
    probability = model.predict_proba(X)[:, 1]

    out = pd.DataFrame(
        {
            "kepoi_name": candidates_df.get("kepoi_name"),
            "kepler_name": candidates_df.get("kepler_name"),
            "kepid": candidates_df["kepid"],
            "probability": probability,
        }
    ).sort_values("probability", ascending=False).head(top_n).reset_index()

    out["rank"] = range(1, len(out) + 1)
    out["top_reasons"] = [
        "; ".join(
            f"{r['feature']}={r['value']:.3g} ({r['shap']:+.3f})"
            for r in explain_row(model, X, index=int(src_idx), top_n=3)
        )
        for src_idx in out["index"]
    ]
    return out.drop(columns=["index"])


def validate_against_snapshot(ranked: pd.DataFrame, newer_koi: pd.DataFrame) -> dict:
    """Check how many ranked candidates a later archive snapshot confirms."""
    status = newer_koi.set_index("kepoi_name")["koi_disposition"]
    checked = ranked[ranked["kepoi_name"].isin(status.index)]
    confirmed = (status.loc[checked["kepoi_name"]] == "CONFIRMED").sum()
    n = int(len(checked))
    return {
        "n_checked": n,
        "n_since_confirmed": int(confirmed),
        "hit_rate": float(confirmed / n) if n else 0.0,
    }
```

```python
from __future__ import annotations

import json
from datetime import UTC, datetime

import joblib
import pandas as pd
import typer

from exodiscover.config import settings
from exodiscover.data import ingest
from exodiscover.data.splits import grouped_train_test_split
from exodiscover.discover import rank_candidates
from exodiscover.evaluate import evaluate_binary, plot_all, reliability_curve, write_metrics
from exodiscover.experiments import ablation, transfer
from exodiscover.explain import global_importance
from exodiscover.features.tabular import FEATURE_COLUMNS, build_features
from exodiscover.models import train as train_mod

app = typer.Typer(help="ExoDiscover — exoplanet classification over NASA catalogs.")


@app.command()
def ingest_cmd(force: bool = False) -> None:
    """Download the Kepler, TESS, and K2 catalogs."""
    for table in ingest.TABLES:
        df = ingest.fetch(table, force=force)
        typer.echo(f"{table}: {len(df)} rows")


app.command("ingest")(ingest_cmd)


@app.command()
def train(fast: bool = typer.Option(False, help="Skip Optuna; for CI smoke runs.")) -> None:
    koi = pd.read_csv(settings.raw_dir / "koi.csv")
    binary = koi[koi["koi_disposition"].isin(["CONFIRMED", "FALSE POSITIVE"])].reset_index(drop=True)
    X = build_features(binary)
    y = (binary["koi_disposition"] == "CONFIRMED").astype(int)
    groups = binary["kepid"]

    ladder = train_mod.fit_candidates(X, y, groups)
    best = ladder[0]
    typer.echo(f"best candidate: {best.name} (PR-AUC {best.cv_scores['pr_auc']:.4f})")

    estimator = best.estimator if fast else train_mod.tune_best(X, y, groups, best.name)
    X_tr, X_te, y_tr, y_te = grouped_train_test_split(X, y, groups)
    model = train_mod.build_calibrated(estimator, X_tr, y_tr, groups.loc[X_tr.index])

    test_metrics = evaluate_binary(model, X_te, y_te)
    y_prob = model.predict_proba(X_te)[:, 1]

    toi_path = settings.raw_dir / "toi.csv"
    transfer_block = (
        transfer.run_transfer(koi, pd.read_csv(toi_path))
        if toi_path.exists()
        else {"in_domain": {}, "zero_shot": {}}
    )

    payload = {
        "model": {
            "name": best.name,
            "version": "0.1.0",
            "trained_at": datetime.now(UTC).isoformat(),
            "framing": "binary",
        },
        "features": FEATURE_COLUMNS,
        "test": test_metrics,
        "cv": best.cv_scores,
        "ladder": [{"name": r.name, **r.cv_scores} for r in ladder],
        "ablation": {
            "leakage": ablation.run_leakage_ablation(koi),
            "framing": ablation.run_framing_ablation(koi),
        },
        "transfer": transfer_block,
        "importance": global_importance(model, X_te),
        "reliability": reliability_curve(y_te.to_numpy(), y_prob),
    }
    write_metrics(payload, settings.root / "docs" / "metrics" / "metrics.json")
    plot_all(y_te, y_prob, settings.root / "docs" / "metrics")

    settings.models_dir.mkdir(parents=True, exist_ok=True)
    joblib.dump({"model": model, "features": FEATURE_COLUMNS}, settings.models_dir / "model.joblib")

    ranked = rank_candidates(model, koi, top_n=50)
    ranked.to_csv(settings.root / "docs" / "metrics" / "top_candidates.csv", index=False)
    typer.echo(f"wrote {len(ranked)} ranked candidates")


@app.command()
def eval() -> None:
    """Print the stored metrics."""
    path = settings.root / "docs" / "metrics" / "metrics.json"
    typer.echo(json.dumps(json.loads(path.read_text())["test"], indent=2))


@app.command()
def predict(csv: str = typer.Argument(..., help="CSV of KOI rows to score.")) -> None:
    bundle = joblib.load(settings.models_dir / "model.joblib")
    X = build_features(pd.read_csv(csv))[bundle["features"]]
    for i, p in enumerate(bundle["model"].predict_proba(X)[:, 1]):
        typer.echo(f"row {i}: planet probability {p:.4f}")
```
